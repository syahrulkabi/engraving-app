Cloudflare Worker Sample Code

                
export default {
  async fetch(request, env) {
    if (request.method === "POST") {
      const jobData = await request.json();
      await env.LASER_JOBS.put(jobData.jobId, JSON.stringify(jobData));
      return new Response(JSON.stringify({ status: "success", id: jobData.jobId }), {
        headers: { "Content-Type": "application/json" }
      });
    }
  }
};
